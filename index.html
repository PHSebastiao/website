// Twitch chat command handler
export const commandHandler = async (
  channel: string,
  username: string,
  text: string,
  msg: CommandMessage
) => {
  const [cmd, ...args] = text.trim().split(" ");
  
  if (isDuplicateMessage(msg.id)) return;
  
  const command = globalCommands[cmd];
  if (!command) return;
  
  if (checkSpamPatterns(text)) {
    await banAs(asUser, channel, msg.userId, BanReason.SPAM);
    return;
  }
  
  await command.execute(channel, username, args, msg);
};
